!
! ============================================================
! Routeur-Lyon-1
! Site : Lyon
! Rôle : Routeur principal - Tunnel VPN primaire
! ============================================================
!

hostname Routeur-Lyon-1
!
no ip domain-lookup
ip domain-name nordtech.local
!
enable secret 123Soleil!
service password-encryption
!
banner motd #
------- ROUTEUR-LYON-1 -------
#
!

! ============================================================
! 1. COMPTES ET AAA
! ============================================================
username admin privilege 15 secret 123Soleil!

tacacs-server host 10.10.110.6
tacacs-server key Cisco123

aaa new-model
aaa authentication login default group tacacs+ local
aaa authorization exec default group tacacs+ local

ip ssh version 2

! ============================================================
! 2. ACL DE GESTION
! ============================================================
ip access-list standard MGMT-ONLY
 permit 10.10.99.0 0.0.0.255
 permit 10.10.40.0 0.0.0.255
 permit 10.20.40.0 0.0.0.255
 permit 10.20.99.0 0.0.0.255
!

! ============================================================
! 3. LIGNES D'ACCÈS
! ============================================================
line console 0
 password 123Soleil!
 login
 exec-timeout 15 0
 logging synchronous
!
line vty 0 4
 access-class MGMT-ONLY in
 login authentication default
 transport input ssh
 exec-timeout 15 0
!

! ============================================================
! 4. INTERFACES
! ============================================================
interface GigabitEthernet0/0
 description Vers-Nuage-WAN-Lien-Primaire
 ip address 100.64.0.2 255.255.255.252
 crypto map CMAP-VPN
 no shutdown
!
interface GigabitEthernet0/1
 description Vers-Switch-Lyon-3-Primaire
 ip address 10.20.253.1 255.255.255.252
 no shutdown
!
interface GigabitEthernet0/2
 description Vers-Switch-Lyon-4-Secours
 ip address 10.20.253.5 255.255.255.252
 no shutdown
!

! ============================================================
! 5. ROUTAGE
! ============================================================
ip route 10.20.0.0 255.255.0.0 10.20.253.2 10
ip route 10.20.0.0 255.255.0.0 10.20.253.6 20
ip route 10.10.0.0 255.255.0.0 100.64.0.1 10

! ============================================================
! 6. VPN IPsec (Tunnel primaire vers Brest-1)
! ============================================================
crypto isakmp policy 10
 encryption aes 256
 hash sha
 authentication pre-share
 group 2
 lifetime 86400
!
crypto isakmp key NordTechBrest/Lyon1! address 100.64.0.1
!
crypto ipsec transform-set TS-VPN esp-aes 256 esp-sha-hmac
!
ip access-list extended VPN-TRAFFIC
 permit ip 10.20.0.0 0.0.255.255 10.10.0.0 0.0.255.255
!
crypto map CMAP-VPN 10 ipsec-isakmp
 set peer 100.64.0.1
 set transform-set TS-VPN
 match address VPN-TRAFFIC
!
end
